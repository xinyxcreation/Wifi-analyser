/* ===================================================== ETAT GLOBAL ===================================================== */ let running = false; let analyzerRunning = false; let analyzerTimer = null; /* * Distance actuelle. */ let distance = 20; /* * Sens : * * -1 = rapprochement * +1 = éloignement */ let movementDirection = -1; /* * Seuil de perte de signal. * * Tiré aléatoirement entre 18 et 22 m. */ let signalLossDistance = 20; /* * Seuil d'approche. * * Tiré aléatoirement entre 4.5 et 5.5 m. */ let closeDistance = 5; /* * Vitesse du déplacement. */ const ANALYZER_INTERVAL = 700; /* * Variation par étape. * * Petite pour garder un déplacement lent. */ const MIN_STEP = 0.15; const MAX_STEP = 0.40; /* ===================================================== NOUVEAU CYCLE ===================================================== */ function createNewCycle() { /* * Nouveau seuil de perte. * * Entre 18 et 22 m. */ signalLossDistance = 18 + Math.random() * 4; /* * Nouveau seuil d'approche. * * Entre 4.5 et 5.5 m. */ closeDistance = 4.5 + Math.random(); } /* ===================================================== NOUVELLE SESSION ===================================================== */ function startNewSession() { running = true; analyzerRunning = true; /* * On commence autour de 20 m. */ distance = 20; /* * Premier mouvement : * vers la cible. */ movementDirection = -1; /* * Génère les seuils. */ createNewCycle(); /* * Etat interface. */ document .getElementById("scanState") .innerHTML = ` RECHERCHE `; document .getElementById("networkState") .textContent = "RÉCEPTION ACTIVE"; document .getElementById("networkState") .className = "info-value state"; /* * Première mesure. */ updateSignal(); updateDirectionDisplay(); /* * Démarre le déplacement. */ startAnalyzerTimer(); } /* ===================================================== ANALYZER ===================================================== */ function analyzerClick() { /* * Premier clic : * démarrage complet. */ if (!running) { startNewSession(); return; } /* * Si on était arrivé au seuil * de rapprochement ou de perte, * on inverse le sens. */ movementDirection *= -1; /* * Nouveau cycle avec * de nouveaux seuils aléatoires. */ createNewCycle(); /* * Relance le mouvement. */ analyzerRunning = true; startAnalyzerTimer(); updateDirectionDisplay(); } /* ===================================================== TIMER ===================================================== */ function startAnalyzerTimer() { stopAnalyzerTimer(); analyzerTimer = null; } } /* ===================================================== TICK ANALYZER ===================================================== */ function analyzerTick() { if (!running) { stopAnalyzerTimer(); return; } /* * Variation lente et aléatoire. */ const variation = MIN_STEP + Math.random() * ( MAX_STEP - MIN_STEP ); /* * Déplacement. */ distance += movementDirection * variation; /* ================================================= RAPPROCHEMENT ================================================= */ if ( movementDirection < 0 ) { /* * Arrivée dans la zone * 4.5 → 5.5 m. */ if ( distance <= closeDistance ) { /* * On place précisément * la distance au seuil. */ distance = closeDistance; /* * Arrêt immédiat. */ analyzerRunning = false; stopAnalyzerTimer(); updateSignal(); updateDirectionDisplay(); addHistory(); return; } } /* ================================================= ELOIGNEMENT ================================================= */ if ( movementDirection > 0 ) { /* * Arrivée au seuil de perte. * * IMPORTANT : * on ne continue PAS jusqu'à 22 m. */ if ( distance >= signalLossDistance ) { /* * Distance exacte de perte. */ distance = signalLossDistance; /* * Arrêt immédiat. */ analyzerRunning = false; stopAnalyzerTimer(); /* * Affiche immédiatement * la perte de signal. */ updateSignal(); updateDirectionDisplay(); addHistory(); return; } } /* * Mise à jour normale. */ updateSignal(); updateDirectionDisplay(); addHistory(); } /* ===================================================== CALCUL RSSI ===================================================== */ function calculateRSSI( currentDistance ) { /* * Valeurs simulées : * * 5 m ≈ -40 dBm * 10 m ≈ -47 dBm * 15 m ≈ -54 dBm * 20 m ≈ -60 dBm */ const base = -34 - ( currentDistance * 1.3 ); return Math.round( base ); } /* ===================================================== UPDATE SIGNAL ===================================================== */ function updateSignal() { /* * Perte uniquement lorsque * la distance atteint le seuil * aléatoire du cycle. */ const signalLost = distance >= signalLossDistance; /* ================================================= PERTE SIGNAL ================================================= */ if (signalLost) { setDeviceLost(1); setDeviceLost(2); document .getElementById("networkState") .textContent = "SIGNAL PERDU"; document .getElementById("networkState") .className = "info-value state lost"; document .getElementById("convergence") .textContent = "PERTE SIGNAL"; document .getElementById("convergence") .classList.add( "lost" ); } else { /* ============================================= RECEPTION ============================================= */ setDeviceReceiving( 1, true ); setDeviceReceiving( 2, true ); /* * RSSI. */ const baseRSSI = calculateRSSI( distance ); /* * Petite variation indépendante. */ const noise1 = Math.round( Math.random() * 4 - 2 ); const noise2 = Math.round( Math.random() * 4 - 2 ); const rssi1 = baseRSSI + noise1; const rssi2 = baseRSSI + noise2; document .getElementById("rssi1") .textContent = rssi1 + " dBm"; document .getElementById("rssi2") .textContent = rssi2 + " dBm"; /* * Barre volontairement courte. */ const signalStrength = 8 + ( (20 - distance) / 15 ) * 27; const barWidth = Math.max( 8, Math.min( 35, signalStrength ) ); document .getElementById("bar1") .style.width = barWidth + "%"; document .getElementById("bar2") .style.width = barWidth + "%"; document .getElementById("bar1") .classList.remove( "lost" ); document .getElementById("bar2") .classList.remove( "lost" ); document .getElementById("networkState") .textContent = "RÉCEPTION ACTIVE"; document .getElementById("networkState") .className = "info-value state"; const convergence = document.getElementById( "convergence" ); convergence.classList.remove( "lost" ); convergence.textContent = analyzerRunning ? "ANALYSE" : "PRÊT"; } /* * Distance affichée. */ const displayDistance = distance.toFixed(1); document .getElementById("distance1") .textContent = "Distance : " + displayDistance + " m"; document .getElementById("distance2") .textContent = "Distance : " + displayDistance + " m"; document .getElementById("distance") .textContent = displayDistance + " m"; } /* ===================================================== APPAREIL EN RECEPTION ===================================================== */ function setDeviceReceiving( number, active ) { const device = document.getElementById( "device" + number ); const status = document.getElementById( "status" + number ); const rssi = document.getElementById( "rssi" + number ); const bar = document.getElementById( "bar" + number ); if (active) { device.classList.remove( "lost" ); device.classList.add( "receiving" ); status.classList.remove( "lost" ); status.classList.add( "receiving" ); status.textContent = "RÉCEPTION"; rssi.classList.remove( "lost" ); rssi.classList.add( "active" ); bar.classList.remove( "lost" ); } } /* ===================================================== APPAREIL PERDU ===================================================== */ function setDeviceLost( number ) { const device = document.getElementById( "device" + number ); const status = document.getElementById( "status" + number ); const rssi = document.getElementById( "rssi" + number ); const bar = document.getElementById( "bar" + number ); device.classList.remove( "receiving" ); device.classList.add( "lost" ); status.classList.remove( "receiving" ); status.classList.add( "lost" ); status.textContent = "PERTE SIGNAL"; rssi.classList.remove( "active" ); rssi.classList.add( "lost" ); rssi.textContent = "0 réception"; bar.classList.add( "lost" ); bar.style.width = "0%"; } /* ===================================================== DIRECTION ===================================================== */ function updateDirectionDisplay() { const element = document.getElementById( "direction" ); if ( !analyzerRunning ) { if ( movementDirection < 0 ) { element.textContent = closeDistance.toFixed(1) + " m — ARRÊT"; } else { element.textContent = signalLossDistance.toFixed(1) + " m — PERTE"; } return; } if ( movementDirection < 0 ) { element.textContent = "← PLUS PROCHE"; } else { element.textContent = "→ PLUS LOIN"; } } /* ===================================================== HISTORIQUE ===================================================== */ function addHistory() { const body = document.getElementById( "history" ); const row = document.createElement( "tr" ); /* * Perte de signal. */ if ( distance >= signalLossDistance ) { row.innerHTML = ` ${getTime()} E8:A0:CD:6D:54:64 0 réception C8:91:43:B9:CA:FC 0 réception ${distance.toFixed(1)} m PERTE SIGNAL `; } else { const baseRSSI = calculateRSSI( distance ); const rssi1 = baseRSSI + Math.round( Math.random() * 4 - 2 ); const rssi2 = baseRSSI + Math.round( Math.random() * 4 - 2 ); row.innerHTML = ` ${getTime()} E8:A0:CD:6D:54:64 ${rssi1} dBm C8:91:43:B9:CA:FC ${rssi2} dBm ${distance.toFixed(1)} m RÉCEPTION `; } body.prepend( row ); while ( body.children.length > 60 ) { body.removeChild( body.lastChild ); } } /* ===================================================== ARRET COMPLET ===================================================== */ function stopSimulation() { stopAnalyzerTimer(); running = false; analyzerRunning = false; distance = 20; movementDirection = -1; signalLossDistance = 20; closeDistance = 5; setDeviceNeutral(1); setDeviceNeutral(2); document .getElementById("rssi1") .textContent = "0 réception"; document .getElementById("rssi2") .textContent = "0 réception"; document .getElementById("bar1") .style.width = "0%"; document .getElementById("bar2") .style.width = "0%"; document .getElementById("bar1") .classList.remove( "lost" ); document .getElementById("bar2") .classList.remove( "lost" ); document .getElementById("distance1") .textContent = "Distance : --"; document .getElementById("distance2") .textContent = "Distance : --"; document .getElementById("distance") .textContent = "-- m"; document .getElementById("direction") .textContent = "--"; document .getElementById("convergence") .textContent = "--"; document .getElementById("convergence") .classList.remove( "lost" ); document .getElementById("networkState") .textContent = "EN ATTENTE"; document .getElementById("networkState") .className = "info-value state off"; document .getElementById("scanState") .innerHTML = ` ARRÊTÉ `; } /* ===================================================== APPAREIL NEUTRE ===================================================== */ function setDeviceNeutral( number ) { const device = document.getElementById( "device" + number ); const status = document.getElementById( "status" + number ); const rssi = document.getElementById( "rssi" + number ); device.classList.remove( "receiving" ); device.classList.remove( "lost" ); status.classList.remove( "receiving" ); status.classList.remove( "lost" ); status.textContent = "NON CONNECTÉ"; rssi.classList.remove( "active" ); rssi.classList.remove( "lost" ); } /* ===================================================== HEURE ===================================================== */ function getTime() { return new Date() .toLocaleTimeString( "fr-FR", { hour: "2-digit", minute: "2-digit", second: "2-digit" } ); } /* ===================================================== INITIALISATION ===================================================== */ window.addEventListener( "DOMContentLoaded", () => { running = false; analyzerRunning = false; distance = 20; movementDirection = -1; signalLossDistance = 20; closeDistance = 5; setDeviceNeutral(1); setDeviceNeutral(2); document .getElementById("rssi1") .textContent = "0 réception"; document .getElementById("rssi2") .textContent = "0 réception"; } );
